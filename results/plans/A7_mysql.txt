-- A7 (mysql) set difference (anti-join vs. negated pattern)

SELECT DISTINCT s.name AS name
FROM edges e
JOIN nodes s ON s.node_id = e.subject_id
JOIN nodes a ON a.node_id = e.object_id
WHERE a.name = 'human' AND e.relation_id = 2
  AND NOT EXISTS (
      SELECT 1 FROM edges e2
      JOIN nodes b ON b.node_id = e2.object_id
      WHERE e2.subject_id = e.subject_id AND e2.relation_id = 2
        AND b.name = 'body')
ORDER BY name
