-- B1k4 (mysql) 4-hop expansion (index-join chain vs. traversal)

SELECT DISTINCT x.name AS name
FROM nodes n
JOIN edges e1 ON e1.subject_id = n.node_id
JOIN edges e2 ON e2.subject_id = e1.object_id
JOIN edges e3 ON e3.subject_id = e2.object_id
JOIN edges e4 ON e4.subject_id = e3.object_id
JOIN nodes x ON x.node_id = e4.object_id
WHERE n.name = 'enzyme' AND e1.weight >= 0.5 AND e2.weight >= 0.5 AND e3.weight >= 0.5 AND e4.weight >= 0.5
ORDER BY name LIMIT 100
